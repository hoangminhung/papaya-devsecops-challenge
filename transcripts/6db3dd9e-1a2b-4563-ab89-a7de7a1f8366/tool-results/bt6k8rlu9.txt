     1	"""Find keys in a JSON document that its schema does not define, by path.
     2	
     3	A JSON Schema without `additionalProperties: false` accepts any extra key, so a
     4	misspelled key (e.g. `require_last_pusher_approval`) passes validation. This helper
     5	walks the document next to the schema and reports every key that is not in the
     6	`properties` of the schema node at that position.
     7	
     8	For `oneOf` / `anyOf` it uses the first branch the value validates against, which
     9	for GitHub rule objects is the branch whose `type` enum matches.
    10	"""
    11	import jsonschema
    12	
    13	
    14	def _branch(value, schema):
    15	    for key in ("oneOf", "anyOf"):
    16	        if key in schema:
    17	            for sub in schema[key]:
    18	                if jsonschema.Draft7Validator(sub).is_valid(value):
    19	                    return _branch(value, sub)
    20	    return schema
    21	
    22	
    23	def unknown_keys(value, schema, path="$", allow=()):
    24	    """Return a list of JSON paths whose key is not defined by the schema.
    25	
    26	    `allow` lists extra paths (e.g. "$.permissions.defaultMode") known from
    27	    documentation but missing from the schema; each one must be cited by the caller.
    28	    """
    29	    found = []
    30	    schema = _branch(value, schema)
    31	    if isinstance(value, dict):
    32	        props = schema.get("properties")
    33	        extra = schema.get("additionalProperties")
    34	        for k, v in value.items():
    35	            p = f"{path}.{k}"
    36	            if props is not None and k in props:
    37	                found += unknown_keys(v, props[k], p, allow)
    38	            elif p in allow:
    39	                continue
    40	            elif isinstance(extra, dict):
    41	                found += unknown_keys(v, extra, p, allow)
    42	            else:
    43	                found.append(p)
    44	    elif isinstance(value, list) and isinstance(schema.get("items"), dict):
    45	        for i, v in enumerate(value):
    46	            found += unknown_keys(v, schema["items"], f"{path}[{i}]", allow)
    47	    return found

    67	def pattern_to_regex(pattern):
    68	    anchored = pattern.startswith("/")
    69	    p = pattern.lstrip("/")
    70	    directory = p.endswith("/")
    71	    p = p.rstrip("/")
    72	    if "/" in p:
    73	        anchored = True  # gitignore: a slash in the middle anchors the pattern
    74	    out = ""
    75	    i = 0
    76	    while i < len(p):
    77	        if p.startswith("**/", i):
    78	            out += "(?:.*/)?"
    79	            i += 3
    80	        elif p.startswith("**", i):
    81	            out += ".*"
    82	            i += 2
    83	        elif p[i] == "*":
    84	            out += "[^/]*"
    85	            i += 1
    86	        elif p[i] == "?":
    87	            out += "[^/]"
    88	            i += 1
    89	        else:
    90	            out += re.escape(p[i])
    91	            i += 1
    92	    prefix = "^" if anchored else "^(?:.*/)?"
    93	    # A literal name (e.g. `/apps/github`, `**/logs`) also matches a directory of that
    94	    # name and its contents; a directory pattern matches only contents; a last segment
    95	    # with a wildcard (`docs/*`) does not descend further (GitHub docs example).
    96	    last = p.rsplit("/", 1)[-1]
    97	    if directory:
    98	        suffix = "/.*$"
    99	    elif "*" in last or "?" in last:
   100	        suffix = "$"
   101	    else:
   102	        suffix = "(?:/.*)?$"
   103	    return re.compile(prefix + out + suffix)

   106	def parse(path):
   107	    rules = []
   108	    with open(path) as f:
   109	        for n, line in enumerate(f, 1):
   110	            line = line.split(" #")[0].strip()
   111	            if not line or line.startswith("#"):
   112	                continue
   113	            parts = line.split()
   114	            rules.append((n, parts[0], parts[1:]))
   115	    return rules
   116	
   117	
   118	def owners_of(rules, path):
   119	    result = None
   120	    for _, pattern, owners in rules:
   121	        if pattern_to_regex(pattern).match(path):
   122	            result = owners
   123	    return result

     1	#!/usr/bin/env python3
     2	"""Extract request-body schemas from GitHub's REST OpenAPI description.
     3	
     4	Usage:
     5	  python3 extract_github_schemas.py <api.github.com.json> <spec-commit-sha> <out-dir>
     6	
     7	The spec is github/rest-api-description, descriptions/api.github.com/api.github.com.json.
     8	It is OpenAPI 3.0: schemas use `nullable` and `$ref`. This script resolves every
     9	`$ref` inline and turns `nullable: true` into a JSON Schema type union with "null",
    10	so Python `jsonschema` (Draft 7) can use the result offline.
    11	
    12	The extracted schemas do NOT set `additionalProperties: false` where GitHub does not,
    13	so a schema check alone does not catch misspelled keys. See tests/schema_keys.py.
    14	"""
    15	import copy
    16	import json
    17	import sys
    18	
    19	BODIES = {
    20	    "github-ruleset-create.json": ("/repos/{owner}/{repo}/rulesets", "post"),
    21	    "github-environment-put.json": ("/repos/{owner}/{repo}/environments/{environment_name}", "put"),
    22	    "github-deployment-branch-policy-post.json": (
    23	        "/repos/{owner}/{repo}/environments/{environment_name}/deployment-branch-policies",
    24	        "post",
    25	    ),
    26	}
    27	
    28	
    29	def resolve(node, spec, depth=0):
    30	    if depth > 40:
    31	        raise RecursionError("schema nesting too deep; recursive $ref?")
    32	    if isinstance(node, list):
    33	        return [resolve(x, spec, depth + 1) for x in node]
    34	    if not isinstance(node, dict):
    35	        return node
    36	    if "$ref" in node:
    37	        path = node["$ref"].lstrip("#/").split("/")
    38	        target = spec
    39	        for part in path:
    40	            target = target[part]
    41	        merged = copy.deepcopy(target)
    42	        merged.update({k: v for k, v in node.items() if k != "$ref"})
    43	        return resolve(merged, spec, depth + 1)
    44	    out = {k: resolve(v, spec, depth + 1) for k, v in node.items()}
    45	    if out.pop("nullable", False):
    46	        t = out.get("type")
    47	        if isinstance(t, str):
    48	            out["type"] = [t, "null"]
    49	        elif t is None and "enum" not in out:
    50	            out = {"anyOf": [out, {"type": "null"}]}
    51	        if "enum" in out and None not in out["enum"]:
    52	            out["enum"] = out["enum"] + [None]
    53	    # OpenAPI-only keywords that Draft 7 does not know are harmless; drop the noisy ones.
    54	    for k in ("example", "examples", "x-github"):
    55	        out.pop(k, None)
    56	    return out
    57	
    58	
    59	def main():
    60	    spec_path, sha, out_dir = sys.argv[1:4]
    61	    with open(spec_path) as f:
    62	        spec = json.load(f)
    63	    for name, (path, method) in BODIES.items():
    64	        body = spec["paths"][path][method]["requestBody"]["content"]["application/json"]["schema"]
    65	        schema = resolve(body, spec)
    66	        schema["$schema"] = "http://json-schema.org/draft-07/schema#"
    67	        schema["$comment"] = (
    68	            f"Extracted from github/rest-api-description@{sha} "
    69	            f"(OpenAPI {spec['openapi']}, API {spec['info']['version']}), "
    70	            f"{method.upper()} {path}, by examples/tools/extract_github_schemas.py"
    71	        )
    72	        with open(f"{out_dir}/{name}", "w") as f:
    73	            json.dump(schema, f, indent=1, sort_keys=True)
    74	            f.write("\n")
    75	        print(f"wrote {out_dir}/{name}")
